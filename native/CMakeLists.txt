cmake_minimum_required(VERSION 3.20)
project(ps5bridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.deps")
set(CHIAKI "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/chiaki/source")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
add_compile_definitions(_WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN CHIAKI_LIB_ENABLE_MBEDTLS
    CHIAKI_VERSION_MAJOR=1 CHIAKI_VERSION_MINOR=10 CHIAKI_VERSION_PATCH=0 CHIAKI_VERSION="1.10.0")
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls" deps/mbedtls EXCLUDE_FROM_ALL)
set(DISABLE_EXTRA_LIBS ON CACHE BOOL "" FORCE)
set(DISABLE_BSON_OUTPUT ON CACHE BOOL "" FORCE)
set(DISABLE_WERROR ON CACHE BOOL "" FORCE)
set(BUILD_APPS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/json-c" deps/json-c EXCLUDE_FROM_ALL)
set(UPNPC_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(UPNPC_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
set(UPNPC_NO_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/miniupnp/miniupnpc" deps/miniupnpc EXCLUDE_FROM_ALL)
set(EVENT__DISABLE_OPENSSL ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_SAMPLES ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_BENCHMARK ON CACHE BOOL "" FORCE)
set(EVENT__LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory("${DEPS}/libevent" deps/libevent EXCLUDE_FROM_ALL)
set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
set(CURL_USE_GSSAPI OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(ENABLE_WEBSOCKETS ON CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
add_subdirectory("${CHIAKI}/third-party/curl" deps/curl EXCLUDE_FROM_ALL)
set(nanopb_BUILD_GENERATOR OFF CACHE BOOL "" FORCE)
set(nanopb_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(nanopb_BUILD_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory("${CHIAKI}/third-party/nanopb" deps/nanopb EXCLUDE_FROM_ALL)
add_library(nanopb INTERFACE)
target_link_libraries(nanopb INTERFACE protobuf-nanopb-static)
target_include_directories(nanopb INTERFACE "${CHIAKI}/third-party/nanopb")
target_compile_definitions(nanopb INTERFACE PB_C99_STATIC_ASSERT)
add_library(Nanopb::nanopb ALIAS nanopb)
file(GLOB GF_SOURCES "${CHIAKI}/third-party/gf-complete/src/gf*.c")
list(FILTER GF_SOURCES EXCLUDE REGEX "test|unit")
add_library(gf_complete STATIC ${GF_SOURCES})
target_include_directories(gf_complete PUBLIC "${CHIAKI}/third-party/gf-complete/include")
set(JERASURE_SOURCES galois jerasure reed_sol cauchy liberation)
foreach(SOURCE IN LISTS JERASURE_SOURCES)
    list(APPEND JERASURE_FILES "${CHIAKI}/third-party/jerasure/src/${SOURCE}.c")
endforeach()
add_library(jerasure STATIC ${JERASURE_FILES})
target_include_directories(jerasure PUBLIC "${CHIAKI}/third-party/jerasure/include")
target_link_libraries(jerasure PUBLIC gf_complete)
add_library(Jerasure::Jerasure ALIAS jerasure)
set(CHIAKI_LIB_ENABLE_OPUS OFF)
set(CHIAKI_ENABLE_FFMPEG_DECODER OFF)
set(CHIAKI_ENABLE_PI_DECODER OFF)
set(CHIAKI_ENABLE_TESTS OFF)
set(CHIAKI_LIB_ENABLE_MBEDTLS ON)
set(CHIAKI_LIB_MBEDTLS_EXTERNAL_PROJECT ON)
set(CHIAKI_LIB_JSONC_EXTERNAL_PROJECT ON)
set(CHIAKI_LIB_MINIUPNPC_EXTERNAL_PROJECT ON)
set(JSONC_PREFIX_INCLUDE_DIR "${DEPS}/prefix")
set(MINIUPNPC_PREFIX_INCLUDE_DIR "${DEPS}/prefix")
set(json-c_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/deps/json-c")
set(PYTHON_EXECUTABLE "${DEPS}/python/Scripts/python.exe")
set(NANOPB_GENERATOR_PY "${CHIAKI}/third-party/nanopb/generator/nanopb_generator.py")
add_subdirectory("${CHIAKI}/lib" chiaki-lib)
add_library(ps5bridge SHARED src/ps5bridge.c)
target_include_directories(ps5bridge PUBLIC include)
target_link_libraries(ps5bridge PRIVATE chiaki-lib winmm setupapi hid)
target_compile_options(ps5bridge PRIVATE -Wall -Wextra -Werror)
set_target_properties(ps5bridge PROPERTIES PREFIX "" OUTPUT_NAME "ps5bridge")
add_executable(ps5bridge-tests tests/test_main.c)
target_link_libraries(ps5bridge-tests PRIVATE ps5bridge)
add_executable(ps5bridge-feedback-tests tests/test_feedback_sender.c
    "${CHIAKI}/lib/src/feedbacksender.c" "${CHIAKI}/lib/src/feedback.c"
    "${CHIAKI}/lib/src/controller.c" "${CHIAKI}/lib/src/thread.c"
    "${CHIAKI}/lib/src/time.c" "${CHIAKI}/lib/src/log.c")
target_include_directories(ps5bridge-feedback-tests PRIVATE "${CHIAKI}/lib/include"
    "${CMAKE_CURRENT_BINARY_DIR}/chiaki-lib/include")
target_link_libraries(ps5bridge-feedback-tests PRIVATE ws2_32 winmm)
set_source_files_properties(tests/test_feedback_sender.c "${CHIAKI}/lib/src/feedbacksender.c"
    PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
add_executable(ps5bridge-input-tests tests/test_input_worker.c)
target_include_directories(ps5bridge-input-tests PRIVATE include)
target_link_libraries(ps5bridge-input-tests PRIVATE chiaki-lib winmm setupapi hid)
target_compile_options(ps5bridge-input-tests PRIVATE -Wall -Wextra -Werror)
